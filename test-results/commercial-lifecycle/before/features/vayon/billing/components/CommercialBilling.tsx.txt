import { BillingPortalButton } from "./StripeBilling";
import type { BillingEventRecord, PaymentMethodRecord } from "../types";
const card="rounded-3xl border border-vds-border bg-vds-surface p-5";
export function PaymentMethodList({items}:{items:PaymentMethodRecord[]}){return <section className="mt-7 space-y-3">{items.map(item=><article className={card} key={item.id}><div className="flex items-center justify-between"><div><p className="font-medium capitalize">{item.brand??item.type} •••• {item.last4??"—"}</p><p className="mt-1 text-xs text-vds-muted">{item.expiry?`Expires ${item.expiry}`:"Expiry unavailable"}</p></div>{item.isDefault&&<span className="text-xs text-vds-primary">Default</span>}</div></article>)}{!items.length&&<div className={card}><p className="text-sm text-vds-muted">No payment method is attached to this Stripe customer.</p></div>}<BillingPortalButton/></section>}
export function BillingHistory({items}:{items:BillingEventRecord[]}){return <section className={`${card} mt-7`}><h2 className="font-semibold">Billing history</h2><div className="mt-4 space-y-3">{items.map(item=><div className="flex justify-between gap-4 border-t border-vds-border pt-3 text-sm" key={item.id}><span className="capitalize">{item.type.replaceAll("."," ")}</span><span className="text-vds-muted">{item.status} · {new Date(item.createdAt).toLocaleString()}</span></div>)}{!items.length&&<p className="text-sm text-vds-muted">No Stripe billing events have been recorded.</p>}</div></section>}
