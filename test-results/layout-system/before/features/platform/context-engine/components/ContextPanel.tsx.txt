"use client";
import { Button } from "@/features/platform/design-system";
import { useMemo, useState } from "react";
import { Boxes, Layers3 } from "lucide-react";
import type { ContextTabId } from "../domain/contracts";
import type { UnifiedContextViewModel } from "../view-models/contracts";
import { ContextSection } from "./ContextSection";
export function UnifiedContextPanel({ model, initialTab = "overview", className = "" }: { readonly model: UnifiedContextViewModel; readonly initialTab?: ContextTabId; readonly className?: string }) { const [active, setActive] = useState<ContextTabId>(initialTab); const tab = model.tabs.find(item => item.id === active) ?? model.tabs[0]; const sections = useMemo(() => model.sections.filter(section => tab?.sections.includes(section.id)), [model.sections, tab]); return <section aria-label="Unified context panel" className={`overflow-hidden rounded-3xl border border-vds-border/[.09] bg-vds-surface shadow-2xl shadow-vds-shadow ${className}`}><header className="border-b border-vds-border/[.07] p-5 sm:p-6"><div className="flex items-start gap-3"><span className="grid size-11 place-items-center rounded-2xl border border-vds-accent-border bg-vds-primary/[.07] text-vds-primary">{model.target ? <Boxes className="size-5" aria-hidden="true"/> : <Layers3 className="size-5" aria-hidden="true"/>}</span><div><p className="text-[10px] uppercase tracking-[.17em] text-vds-primary">Unified context</p><h2 className="mt-1 text-lg font-semibold">{model.title}</h2><p className="mt-1 text-xs capitalize text-vds-subtle">{model.subtitle}</p></div></div><nav aria-label="Context tabs" className="mt-5 flex gap-1 overflow-x-auto pb-1">{model.tabs.map(item => <Button variant="control" key={item.id} type="button" aria-selected={active === item.id} role="tab" onClick={() => setActive(item.id)} className={`shrink-0 rounded-lg px-3 py-2 text-xs transition focus-visible:outline-2 focus-visible:outline-vds-focus ${active === item.id ? "bg-vds-primary-soft text-vds-primary" : "text-vds-subtle hover:bg-vds-surface/[.04] hover:text-vds-secondary"}`}>{item.label}</Button>)}</nav></header><div className="grid gap-3 p-4 sm:p-5 lg:grid-cols-2">{sections.map(section => <ContextSection key={section.id} section={section}/>)}</div></section> }
