"use client";
import { Button } from "@/features/platform/design-system";
import{Command,Search,X}from"lucide-react";import{useEffect,useMemo,useRef,useState}from"react";import{LocalRecentSearchStore,LocalUniversalObjectStore}from"../storage/local.store";import{UniversalSearchService,type SearchHit,type SearchScope}from"../services/search.service";import{EntityBadge}from"./EntityBadge";
const scopes:SearchScope[]=["contact","company","property","lead","deal","task","calendar","document","note","tag","ai-memory"];
const recentStore=new LocalRecentSearchStore(),searchService=new UniversalSearchService(new LocalUniversalObjectStore());
export function SearchOverlay(){const[open,setOpen]=useState(false),[query,setQuery]=useState(""),[results,setResults]=useState<readonly SearchHit[]>([]),[recent,setRecent]=useState<readonly string[]>([]),input=useRef<HTMLInputElement>(null),activeScopes=useMemo(()=>scopes,[]);useEffect(()=>{const key=(event:KeyboardEvent)=>{if((event.ctrlKey||event.metaKey)&&event.key.toLowerCase()==="k"){event.preventDefault();setOpen(value=>!value)}if(event.key==="Escape")setOpen(false)};document.addEventListener("keydown",key);return()=>document.removeEventListener("keydown",key)},[]);useEffect(()=>{if(open)queueMicrotask(()=>{setRecent(recentStore.list());input.current?.focus()})},[open]);async function search(value:string){setQuery(value);setResults(await searchService.search(value,activeScopes))}function remember(value:string){recentStore.push(value);setRecent(recentStore.list())}return <>{<Button variant="control" type="button" onClick={()=>setOpen(true)} aria-label="Open universal search" className="fixed bottom-5 right-5 z-40 flex items-center gap-2 rounded-2xl border border-vds-border bg-[var(--vds-color-elevated)] px-4 py-3 text-xs text-vds-muted shadow-2xl hover:border-vds-accent-border"><Search className="size-4 text-vds-primary"/>Search <kbd className="rounded bg-vds-surface/[.06] px-1.5 py-0.5">Ctrl K</kbd></Button>}{open&&<div role="dialog" aria-modal="true" aria-label="Universal search" className="fixed inset-0 z-[100] bg-vds-overlay p-4 backdrop-blur-sm" onMouseDown={event=>{if(event.currentTarget===event.target)setOpen(false)}}><div className="mx-auto mt-[10vh] max-w-2xl overflow-hidden rounded-3xl border border-vds-border bg-[var(--vds-color-surface)] shadow-2xl"><div className="flex items-center gap-3 border-b border-vds-border/[.07] p-4"><Command className="size-5 text-vds-primary"/><input ref={input} value={query} onChange={event=>void search(event.target.value)} onKeyDown={event=>{if(event.key==="Enter")remember(query)}} placeholder="Search every business object…" className="h-10 flex-1 bg-transparent outline-none"/><Button variant="control" type="button" onClick={()=>setOpen(false)} aria-label="Close search"><X className="size-5 text-vds-muted"/></Button></div><div className="flex gap-1 overflow-x-auto border-b border-vds-border/[.06] p-3">{scopes.map(scope=><span key={scope} className="rounded-full bg-vds-surface/[.04] px-2 py-1 text-[9px] uppercase text-vds-subtle">{scope}</span>)}</div><div className="max-h-[55vh] overflow-y-auto p-3">{query&&results.map(result=><Button variant="control" key={`${result.type}-${result.id}`} type="button" onClick={()=>remember(query)} className="flex w-full items-center justify-between rounded-2xl p-3 text-left hover:bg-vds-surface/[.05]"><span><span className="block text-sm">{result.title}</span><span className="mt-1 block text-xs text-vds-subtle">{result.excerpt}</span></span><EntityBadge type={result.type}/></Button>)}{query&&!results.length&&<div className="p-8 text-center"><p className="text-sm text-vds-muted">No locally indexed objects match.</p><p className="mt-2 text-xs text-vds-subtle">The backend search adapter is intentionally a placeholder in Release 0.5.</p></div>}{!query&&<div><p className="px-3 py-2 text-[10px] uppercase tracking-wider text-vds-subtle">Recent searches</p>{recent.map(item=><Button variant="control" key={item} type="button" onClick={()=>void search(item)} className="block w-full rounded-xl px-3 py-2 text-left text-sm text-vds-muted hover:bg-vds-surface/[.04]">{item}</Button>)}{!recent.length&&<p className="p-6 text-center text-xs text-vds-subtle">No recent searches.</p>}</div>}</div></div></div>}</>}
