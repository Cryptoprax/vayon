export function UniversalObjectsWorkbench(){const[tab,setTab]=useState<Tab>("contacts"),[objects,setObjects]=useState<UniversalObject[]>([]),[relationships,setRelationships]=useState<UniversalRelationship[]>([]);useEffect(()=>{queueMicrotask(()=>{setObjects([...service.list()]);setRelationships([...relationshipStore.list()])})},[]);function save(object:UniversalObject){service.save(object);setObjects([...service.list()])}const contacts=objects.filter((item):item is UniversalContact=>item.type==="contact"),companies=objects.filter((item):item is UniversalCompany=>item.type==="company"),activities=objects.filter((item):item is UniversalActivity=>item.type==="activity");return <div><nav className="flex gap-2 overflow-x-auto rounded-2xl border border-vds-border/[.07] bg-vds-surface/[.025] p-2">{tabs.map(item=><Button variant="control" key={item.id} type="button" onClick={()=>setTab(item.id)} className={`flex items-center gap-2 whitespace-nowrap rounded-xl px-3 py-2.5 text-xs ${tab===item.id?"bg-vds-primary-soft text-vds-primary":"text-vds-muted hover:bg-vds-surface/[.05]"}`}><item.icon className="size-4"/>{item.label}</Button>)}</nav><div className="mt-6">{tab==="contacts"&&<ContactCenter items={contacts} companies={companies} onSave={save}/>} {tab==="companies"&&<CompanyCenter items={companies} onSave={save}/>} {tab==="address"&&<AddressCard/>}{tab==="documents"&&<DocumentCenter/>}{tab==="activity"&&<ActivityTimeline items={activities}/>} {tab==="notes"&&<NotesManager/>}{tab==="attachments"&&<AttachmentManager/>}{tab==="tags"&&<TagManager/>}{tab==="relationships"&&<RelationshipGraph relationships={relationships} onAdd={relationship=>{relationshipStore.save(relationship);setRelationships([...relationshipStore.list()])}}/>}</div></div>}
