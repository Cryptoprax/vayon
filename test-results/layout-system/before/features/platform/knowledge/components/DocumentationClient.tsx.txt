function ArticleCard({article}:{article:DocumentationArticle}){return <Link href={`/docs/${article.slug}`} className="vds-focus rounded-2xl border border-vds-border bg-vds-surface p-5 hover:border-vds-accent-border"><p className="text-xs text-vds-primary">{article.category} · v{article.version}</p><h2 className="mt-2 font-semibold">{article.title}</h2><p className="mt-2 text-sm leading-6 text-vds-muted">{article.description}</p><p className="mt-4 text-xs text-vds-subtle">{article.readingMinutes} min read</p></Link>}
function SidebarList({title,articles}:{title:string;articles:readonly DocumentationArticle[]}){return <section><h2 className="text-sm font-semibold">{title}</h2><ul className="mt-3 space-y-2">{articles.map(article=><li key={article.slug}><Link className="text-xs text-vds-muted hover:text-vds-foreground" href={`/docs/${article.slug}`}>{article.title}</Link></li>)}</ul></section>}

export function DocumentationArticleView({article,related}:{article:DocumentationArticle;related:readonly (DocumentationArticle|undefined)[]}){const[copied,setCopied]=useState(""),[bookmarked,setBookmarked]=useState(false),[progress,setProgress]=useState(0);useEffect(()=>{setBookmarked(stored(bookmarkKey).includes(article.slug));const recent=[article.slug,...stored(recentKey).filter(x=>x!==article.slug)].slice(0,8);localStorage.setItem(recentKey,JSON.stringify(recent));void telemetry.record({type:"view",articleSlug:article.slug});const scroll=()=>{const maximum=document.documentElement.scrollHeight-innerHeight;setProgress(maximum>0?Math.min(100,Math.round(scrollY/maximum*100)):100)};scroll();addEventListener("scroll",scroll,{passive:true});return()=>removeEventListener("scroll",scroll)},[article.slug]);function bookmark(){const current=stored(bookmarkKey),next=current.includes(article.slug)?current.filter(x=>x!==article.slug):[...current,article.slug];localStorage.setItem(bookmarkKey,JSON.stringify(next));setBookmarked(next.includes(article.slug));void telemetry.record({type:"bookmark",articleSlug:article.slug})}return <><div className="fixed inset-x-0 top-0 z-50 h-1 bg-vds-elevated" aria-hidden="true"><div className="h-full bg-vds-primary" function Block({block,copied,setCopied}:{block:DocumentationBlock;copied:string;setCopied:(value:string)=>void}){if(block.type==="paragraph")return <p className="leading-7 text-vds-muted">{block.text}</p>;if(block.type==="steps")return <ol className="list-decimal space-y-3 pl-5 text-vds-muted">{block.items.map(item=><li key={item}>{item}</li>)}</ol>;if(block.type==="callout")return <aside className="rounded-2xl border border-vds-accent-border bg-vds-primary-soft p-5"><strong>{block.title}</strong><p className="mt-2 text-sm text-vds-muted">{block.text}</p></aside>;return <div className="overflow-hidden rounded-2xl border border-vds-border bg-vds-elevated"><div className="flex items-center justify-between border-b border-vds-border px-4 py-2 text-xs text-vds-subtle"><span>{block.language}</span><Button variant="control" onClick={async()=>{await navigator.clipboard.writeText(block.code);setCopied(block.code)}}>{copied===block.code?<Check className="size-4"/>:<Copy className="size-4"/>} Copy</Button></div><pre className="overflow-x-auto p-5 text-sm"><code>{block.code}</code></pre></div>}
