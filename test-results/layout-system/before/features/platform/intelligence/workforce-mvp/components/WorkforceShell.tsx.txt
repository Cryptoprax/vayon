"use client";
import { Button } from "@/features/platform/design-system";
import { useMemo, useState } from "react";
import { Activity, ShieldCheck, Sparkles } from "lucide-react";
import { advisorDefinitions } from "../services/advisor.service";
import { emptyAdvisorViewModel } from "../view-models/advisor";
import type { AdvisorType, ConversationMessage, RecommendationTimelineEntry } from "../domain/contracts";
import { AdvisorSelector } from "./AdvisorSelector";
import { ConversationPanel } from "./ConversationPanel";
import { ExplainabilityDrawer } from "./ExplainabilityDrawer";
import { InsightCard } from "./InsightCard";
import { RecommendationTimeline } from "./RecommendationTimeline";

export function WorkforceShell() {
  const [active, setActive] = useState<AdvisorType>("executive"), [drawer, setDrawer] = useState(false), [message, setMessage] = useState(""), [messages, setMessages] = useState<ConversationMessage[]>([]);
  const definition = advisorDefinitions.find(item => item.id === active) ?? advisorDefinitions[0], advisor = useMemo(() => emptyAdvisorViewModel(definition), [definition]);
  const advisorMessages = messages.filter(item => item.advisor === active), timeline: readonly RecommendationTimelineEntry[] = [];
  function submit() { const prompt = message.trim(); if (!prompt) return; const now = new Date().toISOString(), suffix = `${Date.now()}`; setMessages(current => [...current, { id: `user:${suffix}`, advisor: active, role: "user", content: prompt, createdAt: now, status: "complete" }, { id: `advisor:${suffix}`, advisor: active, role: "advisor", content: "Advisory processing is not connected in this architecture preview. No business assessment or action was generated.", createdAt: now, status: "architecture-placeholder" }]); setMessage("") }
  return <><div className="space-y-6"><section className="relative overflow-hidden rounded-[2rem] border border-vds-accent-border bg-gradient-to-br from-vds-primary/[.08] via-[var(--vds-color-surface)] to-vds-accent/[.07] p-6 sm:p-8"><div className="absolute right-8 top-4 size-36 rounded-full bg-vds-primary-soft blur-3xl" aria-hidden="true"/><div className="relative flex flex-col gap-6 lg:flex-row lg:items-end lg:justify-between"><div><div className="flex items-center gap-3"><span className="grid size-12 place-items-center rounded-2xl border border-vds-accent-border bg-vds-primary-soft text-vds-primary"><Sparkles aria-hidden="true"/></span><span className="rounded-full border border-vds-success bg-vds-success/[.07] px-3 py-1 text-xs text-vds-success">Advisory only</span></div><h1 className="mt-6 text-3xl font-semibold tracking-tight sm:text-5xl">Vayon Workforce</h1><p className="mt-3 max-w-3xl text-sm leading-6 text-vds-muted sm:text-base">Executive, Sales, and Operations advisors governed by Vayon Brain and the Cognitive Engine. Insights remain explainable, permission-aware, and non-executable.</p></div><div className="flex items-center gap-2 rounded-2xl border border-vds-border/[.08] bg-vds-input px-4 py-3 text-xs text-vds-muted"><ShieldCheck className="size-4 text-vds-success" aria-hidden="true"/>No automatic actions</div></div></section><AdvisorSelector advisors={advisorDefinitions} active={active} onSelect={setActive}/><section aria-labelledby="insights-title"><div className="flex flex-wrap items-end justify-between gap-3"><div><p className="text-xs uppercase tracking-[.2em] text-vds-primary">{definition.name}</p><h2 id="insights-title" className="mt-2 text-xl font-semibold">Advisory workspace</h2><p className="mt-1 text-sm text-vds-muted">{advisor.summary}</p></div><Button variant="control" type="button" onClick={() => setDrawer(true)} className="focus-ring rounded-xl border border-vds-border bg-vds-surface/[.035] px-4 py-2 text-sm text-vds-secondary hover:border-vds-accent-border hover:text-vds-foreground">Open explainability</Button></div><div className="mt-4 grid gap-3 md:grid-cols-2 xl:grid-cols-3">{advisor.insights.map(insight => <InsightCard key={insight.id} insight={insight} onExplain={() => setDrawer(true)}/>)}</div></section><div className="grid gap-6 xl:grid-cols-[1.25fr_.75fr]"><ConversationPanel messages={advisorMessages} value={message} onChange={setMessage} onSubmit={submit}/><div className="space-y-6"><RecommendationTimeline entries={timeline}/><section className="rounded-3xl border border-vds-border/[.08] bg-vds-surface/[.02] p-5"><div className="flex items-center gap-2"><Activity className="size-4 text-vds-muted" aria-hidden="true"/><h2 className="font-semibold">Activity history</h2></div><div className="mt-4 rounded-2xl border border-dashed border-vds-border/[.08] p-6 text-center text-xs text-vds-subtle">No workforce activity has been recorded.</div></section></div></div></div><ExplainabilityDrawer open={drawer} advisor={advisor} onClose={() => setDrawer(false)}/></>;
}
