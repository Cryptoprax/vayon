"use client";

import { memo, useMemo, useState } from "react";
import { Button } from "@/features/platform/design-system";
import type { ExecutiveAIDashboard } from "@/features/platform/executive-ai/types";
import type { ExecutiveBIModel } from "../domain/models";

const card = "rounded-3xl border border-vds-border bg-vds-surface p-5 shadow-vds-sm";
const money = (value: number | null) => value == null ? "Unavailable" : new Intl.NumberFormat("en-IN", { style: "currency", currency: "INR", maximumFractionDigits: 0 }).format(value);

export const ExecutiveBI = memo(function ExecutiveBI({ data, ai }: { data: ExecutiveBIModel; ai: ExecutiveAIDashboard }) {
  const [range, setRange] = useState("quarter"), [team, setTeam] = useState("all"), [view, setView] = useState("Executive overview");
  const csv = useMemo(() => ["metric,value,available,source", ...data.metrics.map((item) => [item.label, item.value ?? "Unavailable", item.available, item.source].map((value) => `"${String(value).replaceAll('"', '""')}"`).join(","))].join("\n"), [data.metrics]);
  const exportCsv = () => { const link = document.createElement("a"); link.href = URL.createObjectURL(new Blob([csv], { type: "text/csv" })); link.download = `vayon-executive-${range}.csv`; link.click(); URL.revokeObjectURL(link.href); };
  return <div className="space-y-7">
    <section className={`${card} flex flex-col gap-4 lg:flex-row lg:items-end lg:justify-between`} aria-label="Executive report controls"><div className="grid flex-1 gap-3 sm:grid-cols-3"><label className="text-xs text-vds-muted">Date range<select value={range} onChange={(event) => setRange(event.target.value)} className="mt-1 h-10 w-full rounded-xl border border-vds-border bg-vds-input px-3 text-sm text-vds-foreground"><option value="yesterday">Yesterday</option><option value="week">Last week</option><option value="month">Last month</option><option value="quarter">Quarter</option><option value="annual">Annual</option></select></label><label className="text-xs text-vds-muted">Team / department<select value={team} onChange={(event) => setTeam(event.target.value)} className="mt-1 h-10 w-full rounded-xl border border-vds-border bg-vds-input px-3 text-sm text-vds-foreground"><option value="all">All teams</option>{data.benchmarks.map((item) => <option value={item.label} key={item.label}>{item.label}</option>)}</select></label><label className="text-xs text-vds-muted">Saved view<select value={view} onChange={(event) => setView(event.target.value)} className="mt-1 h-10 w-full rounded-xl border border-vds-border bg-vds-input px-3 text-sm text-vds-foreground"><option>Executive overview</option><option>Revenue review</option><option>Sales health</option><option>Operations review</option></select></label></div><div className="flex flex-wrap gap-2"><Button variant="outline" onClick={exportCsv}>Export CSV</Button><Button variant="control" disabled title="Prepared export placeholder">PDF soon</Button><Button variant="control" disabled title="Prepared export placeholder">Excel soon</Button><Button variant="control" disabled title="Prepared export placeholder">Presentation soon</Button></div></section>
    <section className="grid gap-3 sm:grid-cols-2 lg:grid-cols-3 xl:grid-cols-6">{data.metrics.map((item) => <article className={card} key={item.id}><div className="flex items-center justify-between"><p className="text-xs text-vds-muted">{item.label}</p><span className={`size-2 rounded-full ${item.available ? "bg-vds-success" : "bg-vds-warning"}`} aria-label={item.available ? "Measured" : "Unavailable"}/></div><p className="mt-3 text-xl font-semibold">{item.available ? typeof item.value === "number" && ["revenue", "arr", "mrr", "forecast", "pipeline"].includes(item.id) ? money(item.value) : item.value : "Unavailable"}</p><p className="mt-2 line-clamp-2 text-[10px] text-vds-muted">{item.explanation}</p></article>)}</section>
    <section className="grid gap-5 xl:grid-cols-3"><article className={`${card} xl:col-span-2`}><h2 className="font-semibold">Forecast scenarios</h2><div className="mt-5 grid gap-3 sm:grid-cols-3">{[["Best case", data.forecast.bestCase], ["Expected", data.forecast.expected], ["Worst case", data.forecast.worstCase], ["Weighted forecast", data.forecast.weighted], ["Quarter forecast", data.forecast.quarter], ["Annual forecast", data.forecast.annual]].map(([label, value]) => <div className="rounded-2xl bg-vds-elevated p-4" key={label as string}><p className="text-xs text-vds-muted">{label}</p><p className="mt-2 font-semibold">{money(value as number | null)}</p></div>)}</div><div className="mt-5"><div className="flex justify-between text-xs"><span>Forecast confidence</span><span>{data.forecast.confidence}%</span></div><div className="mt-2 h-2 overflow-hidden rounded-full bg-vds-elevated"><span className="block h-full rounded-full bg-vds-primary" style={{ width: `${data.forecast.confidence}%` }}/></div></div></article><article className={card}><h2 className="font-semibold">Executive digest</h2><div className="mt-4 space-y-3">{data.digest.map((item) => <div className="rounded-xl border border-vds-border p-3" key={item.period}><p className="text-sm font-medium">{item.period}</p><p className="mt-1 text-xs text-vds-muted">{item.achievements} achievements · {item.risks} risks · {item.pendingApprovals} approvals</p></div>)}</div></article></section>
    <section className="grid gap-5 xl:grid-cols-2"><BarChart title="Lead sources" items={data.leadSources}/><Funnel title="Pipeline leakage and stage distribution" items={data.stages}/><BarChart title="Top performers · closed wins" items={data.performers}/><Donut title="Lost reasons" items={data.lostReasons}/><TrendChart title="Pipeline trend · line and area" items={data.stages}/><Heatmap title="Stage performance heatmap" items={data.stages}/></section>
    <section className={card}><div className="flex items-center justify-between"><div><p className="text-xs font-semibold uppercase tracking-[.18em] text-vds-primary">Generated insight · recommendation only</p><h2 className="mt-2 font-semibold">Executive AI recommendations</h2></div><span className="rounded-full border border-vds-accent-border px-3 py-1 text-[10px] uppercase text-vds-primary">AI generated</span></div><div className="mt-5 grid gap-3 md:grid-cols-2 xl:grid-cols-3">{ai.priorities.map((item) => <article className="rounded-2xl bg-vds-elevated p-4" key={item.id}><div className="flex justify-between gap-3"><h3 className="font-medium">{item.title}</h3><span className="text-[10px] uppercase">{item.severity}</span></div><p className="mt-2 text-sm text-vds-muted">{item.explanation}</p><p className="mt-3 text-xs text-vds-primary">Recommendation: {item.suggestedAction}</p><p className="mt-2 text-[10px] text-vds-muted">{Math.round(item.confidence * 100)}% confidence · generated, not measured</p></article>)}</div>{!ai.priorities.length && <p className="mt-5 text-sm text-vds-muted">No evidence-backed AI recommendations are available.</p>}</section>
    <section className={card}><h2 className="font-semibold">Internal benchmarking</h2><p className="mt-1 text-xs text-vds-muted">Compare recorded salespeople and teams. Personal rankings reflect tenant-scoped pipeline and closed outcomes only.</p><div className="mt-4 overflow-x-auto"><table className="w-full min-w-[36rem] text-left text-sm"><thead className="text-xs text-vds-muted"><tr><th className="pb-3">Team / salesperson</th><th className="pb-3">Pipeline</th><th className="pb-3">Wins</th><th className="pb-3">Selected</th></tr></thead><tbody>{data.benchmarks.map((item) => <tr className="border-t border-vds-border" key={item.label}><td className="py-3">{item.label}</td><td>{money(item.pipeline)}</td><td>{item.wins}</td><td>{team === "all" || team === item.label ? "Included" : "Filtered"}</td></tr>)}</tbody></table></div></section>
    <p className="text-xs text-vds-muted">View: {view} · Range: {range} · Generated {new Date(data.generatedAt).toLocaleString()} · Source: {data.source}</p>
  </div>;
});

function BarChart({ title, items }: { title: string; items: readonly { label: string; value: number }[] }) { const max = Math.max(1, ...items.map((item) => item.value)); return <article className={card}><h2 className="font-semibold">{title}</h2><div className="mt-5 space-y-3">{items.slice(0, 12).map((item) => <div key={item.label}><div className="flex justify-between text-xs"><span className="truncate">{item.label}</span><span>{item.value}</span></div><div className="mt-1 h-2 rounded-full bg-vds-elevated"><span className="block h-full rounded-full bg-vds-primary" style={{ width: `${item.value / max * 100}%` }}/></div></div>)}{!items.length && <Empty/>}</div></article>; }
function Funnel({ title, items }: { title: string; items: readonly { label: string; value: number; amount: number }[] }) { const max = Math.max(1, ...items.map((item) => item.value)); return <article className={card}><h2 className="font-semibold">{title}</h2><div className="mt-5 space-y-2">{items.map((item) => <div className="mx-auto rounded-lg bg-vds-primary-soft px-3 py-2 text-center text-xs" style={{ width: `${Math.max(32, item.value / max * 100)}%` }} key={item.label}>{item.label} · {item.value} · {money(item.amount)}</div>)}{!items.length && <Empty/>}</div></article>; }
function Donut({ title, items }: { title: string; items: readonly { label: string; value: number }[] }) { const total = items.reduce((sum, item) => sum + item.value, 0); return <article className={card}><h2 className="font-semibold">{title}</h2><div className="mt-5 flex items-center gap-5"><div className="grid size-28 shrink-0 place-items-center rounded-full border-[16px] border-vds-accent-border bg-vds-elevated"><span className="text-lg font-semibold">{total}</span></div><div className="min-w-0 space-y-2">{items.slice(0, 8).map((item) => <p className="flex justify-between gap-4 text-xs" key={item.label}><span className="truncate text-vds-muted">{item.label}</span><span>{item.value}</span></p>)}{!items.length && <Empty/>}</div></div></article>; }
function TrendChart({ title, items }: { title: string; items: readonly { label: string; value: number }[] }) { const max = Math.max(1, ...items.map((item) => item.value)), points = items.slice(0, 10).map((item, index, list) => `${list.length === 1 ? 50 : index / (list.length - 1) * 100},${92 - item.value / max * 78}`).join(" "); return <article className={card}><h2 className="font-semibold">{title}</h2>{items.length ? <><svg viewBox="0 0 100 100" className="mt-5 h-44 w-full" role="img" aria-label="Stage pipeline line and area chart"><defs><linearGradient id="executive-area" x1="0" y1="0" x2="0" y2="1"><stop offset="0" stopColor="var(--vds-color-primary)" stopOpacity=".3"/><stop offset="1" stopColor="var(--vds-color-primary)" stopOpacity="0"/></linearGradient></defs><polygon points={`0,100 ${points} 100,100`} fill="url(#executive-area)"/><polyline points={points} fill="none" stroke="var(--vds-color-primary)" strokeWidth="2" vectorEffect="non-scaling-stroke"/></svg><div className="flex justify-between gap-2 text-[10px] text-vds-muted">{items.slice(0, 5).map((item) => <span className="truncate" key={item.label}>{item.label}</span>)}</div></> : <Empty/>}</article>; }
function Heatmap({ title, items }: { title: string; items: readonly { label: string; value: number }[] }) { const max = Math.max(1, ...items.map((item) => item.value)); return <article className={card}><h2 className="font-semibold">{title}</h2><div className="mt-5 grid grid-cols-2 gap-2 sm:grid-cols-4">{items.slice(0, 12).map((item) => <div className="rounded-xl border border-vds-border p-3" style={{ backgroundColor: `color-mix(in srgb, var(--vds-color-primary) ${Math.max(8, item.value / max * 42)}%, transparent)` }} key={item.label}><p className="truncate text-xs">{item.label}</p><p className="mt-2 font-semibold">{item.value}</p></div>)}{!items.length && <Empty/>}</div></article>; }
function Empty() { return <p className="py-8 text-center text-xs text-vds-muted">No authoritative data for this view.</p>; }
