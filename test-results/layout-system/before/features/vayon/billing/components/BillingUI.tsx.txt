import Link from "next/link";
import { Button } from "@/features/platform/design-system";
import type { BillingContactRecord, InvoiceRecord, PlanRecord, SubscriptionRecord, UsageRecord } from "../types";
import { changePlanAction } from "../actions/billing.actions";
const card="rounded-3xl border border-vds-border bg-vds-surface p-5";
export function BillingHeader({title,description}:{title:string;description:string}){return <header><p className="text-xs uppercase tracking-[.22em] text-vds-primary">Organization billing</p><h1 className="mt-2 text-3xl font-semibold">{title}</h1><p className="mt-2 max-w-2xl text-sm text-vds-muted">{description}</p></header>}
export function SubscriptionStatus({subscription}:{subscription:SubscriptionRecord|null}){return <section className={`${card} mt-7 grid gap-5 sm:grid-cols-2 lg:grid-cols-4`}><Metric label="Current Plan" value={subscription?.planName??"Not activated"}/><Metric label="Status" value={subscription?.status??"Setup required"}/><Metric label="Trial" value={subscription?.trialEndsAt?`Ends ${new Date(subscription.trialEndsAt).toLocaleDateString()}`:"No active trial"}/><Metric label="Renewal" value={subscription?.currentPeriodEndsAt?new Date(subscription.currentPeriodEndsAt).toLocaleDateString():"Unavailable"}/></section>}
export function PlanCard({plan,subscription}:{plan:PlanRecord;subscription:SubscriptionRecord|null}){const active=subscription?.planId===plan.id;return <article className={`${card} ${active?"border-vds-accent-border":""}`}><div className="flex justify-between"><div><h2 className="text-xl font-semibold">{plan.name}</h2><p className="mt-1 text-sm text-vds-muted">{plan.description}</p></div>{active&&<span className="text-xs text-vds-primary">Current</span>}</div><p className="mt-6 text-3xl font-semibold">{plan.monthlyPrice===null?"Custom pricing":new Intl.NumberFormat("en-IN",{style:"currency",currency:plan.currency,maximumFractionDigits:0}).format(plan.monthlyPrice)+"/month"}</p><ul className="mt-5 space-y-2 text-sm text-vds-muted">{plan.features.map(feature=><li key={feature}>✓ {feature}</li>)}</ul>{subscription&&!active&&plan.monthlyPrice!==null&&<form action={changePlanAction} className="mt-6"><input type="hidden" name="planCode" value={plan.code}/><input type="hidden" name="expectedVersion" value={subscription.version}/><input type="hidden" name="seatQuantity" value={subscription.seatQuantity}/><Button variant="control" className="w-full">Change to {plan.name}</Button></form>}</article>}
export function UsageCard({item}:{item:UsageRecord}){const percent=item.limit===null||item.limit===undefined?0:item.limit===0?(item.quantity?100:0):Math.min(100,item.quantity/item.limit*100);return <article className={card}><div className="flex justify-between"><p className="capitalize">{item.metric.replaceAll("_"," ")}</p><p className="text-sm text-vds-muted">{item.quantity} / {item.limit??"∞"}</p></div><div className="mt-4 h-2 overflow-hidden rounded-full bg-vds-elevated"><div className="h-full rounded-full bg-vds-primary" style={{width:`${percent}%`}}/></div></article>}
export function UsageCharts({items}:{items:UsageRecord[]}){return <div className="mt-7 grid gap-3 md:grid-cols-2">{items.map(item=><UsageCard item={item} key={item.metric}/>)}</div>}
export function InvoiceTable({items}:{items:InvoiceRecord[]}){return <div className="mt-7 overflow-x-auto rounded-3xl border border-vds-border"><table className="w-full text-left text-sm"><thead className="bg-vds-elevated text-xs uppercase text-vds-muted"><tr><th className="p-4">Invoice</th><th>Status</th><th>Total</th><th>Issued</th><th>Receipt</th></tr></thead><tbody>{items.map(item=><tr className="border-t border-vds-border" key={item.id}><td className="p-4 font-medium"><Link href={`/vayon/settings/invoices/${item.id}`}>{item.number}</Link></td><td className="capitalize">{item.status}</td><td>{item.currency} {item.total}</td><td>{item.issuedAt?new Date(item.issuedAt).toLocaleDateString():"Pending"}</td><td>{item.downloadUrl?<a className="text-vds-primary" href={item.downloadUrl} target="_blank" rel="noreferrer">Open</a>:"Unavailable"}</td></tr>)}</tbody></table>{!items.length&&<p className="p-12 text-center text-sm text-vds-muted">No Paddle invoices have been recorded.</p>}</div>}
export function SeatManager({subscription}:{subscription:SubscriptionRecord|null}){return <article className={card}><p className="text-xs uppercase text-vds-muted">Users</p><p className="mt-3 text-3xl font-semibold">{subscription?.seatQuantity??0}</p><p className="mt-2 text-sm text-vds-muted">Plan changes are prorated through Paddle.</p></article>}
export function BillingContact({contact}:{contact:BillingContactRecord|null}){return <div className={card}><Metric label="Billing Company" value={contact?.companyName??"Not configured"}/><div className="mt-5"><Metric label="Billing Email" value={contact?.billingEmail??"Not configured"}/></div><div className="mt-5"><Metric label="GST / Tax ID" value={contact?.taxId??"Not provided"}/></div></div>}
function Metric({label,value}:{label:string;value:string}){return <div><p className="text-xs uppercase tracking-wide text-vds-muted">{label}</p><p className="mt-2 font-medium capitalize">{value.replaceAll("_"," ")}</p></div>}
