"use client";
import { Button } from "@/features/platform/design-system";
import { useState } from "react";
import type { ChartPoint } from "../types";
type Metric="revenue"|"pipeline"|"leads"|"conversion"|"sales";
const metrics:{key:Metric;label:string}[]=[{key:"revenue",label:"Revenue"},{key:"pipeline",label:"Pipeline Growth"},{key:"leads",label:"Leads"},{key:"conversion",label:"Conversion Rate"},{key:"sales",label:"Monthly Sales"}];
export default function RevenueChart({data,currency}:{data:ChartPoint[];currency:string}){const[metric,setMetric]=useState<Metric>("revenue"),values=data.map((point)=>point[metric]),max=Math.max(...values,1),points=values.map((value,index)=>`${index/(Math.max(values.length-1,1))*100},${92-value/max*78}`).join(" "),total=values.reduce((sum,value)=>sum+value,0),format=(value:number)=>metric==="revenue"||metric==="pipeline"?new Intl.NumberFormat("en",{style:"currency",currency,notation:"compact",maximumFractionDigits:1}).format(value):metric==="conversion"?`${value}%`:String(value);return <section className="rounded-3xl border border-vds-border/[0.075] bg-vds-surface/[0.025] p-5 sm:p-6" aria-labelledby="revenue-heading"><div className="flex flex-col gap-4 lg:flex-row lg:items-end lg:justify-between"><div><p className="text-xs font-medium uppercase tracking-[.18em] text-vds-success">Analytics</p><h2 id="revenue-heading" className="mt-2 text-xl font-semibold">Revenue & Growth</h2><p className="mt-2 text-3xl font-semibold">{format(total)}</p></div><div className="flex max-w-full gap-1 overflow-x-auto rounded-xl border border-vds-border/[0.06] bg-vds-input p-1" role="tablist" aria-label="Chart metric">{metrics.map((item)=><Button variant="control" type="button" role="tab" aria-selected={metric===item.key} key={item.key} onClick={()=>setMetric(item.key)} className={`shrink-0 rounded-lg px-3 py-2 text-xs ${metric===item.key?"bg-vds-surface/[0.09] text-vds-foreground":"text-vds-muted hover:text-vds-secondary"}`}>{item.label}</Button>)}</div></div><div className="mt-7"><svg viewBox="0 0 100 100" preserveAspectRatio="none" className="h-64 w-full overflow-visible" role="img" aria-label={`${metrics.find((item)=>item.key===metric)?.label} over twelve months`}><defs><linearGradient id="revenue-area" x1="0" y1="0" x2="0" y2="1"><stop offset="0" stopColor="var(--vds-color-primary)" stopOpacity=".22"/><stop offset="1" stopColor="var(--vds-color-primary)" stopOpacity="0"/></linearGradient></defs>{[20,40,60,80].map((y)=><line key={y} x1="0" x2="100" y1={y} y2={y} stroke="var(--vds-color-skeleton)" strokeWidth=".5"/>)}<polygon points={`0,100 ${points} 100,100`} fill="url(#revenue-area)"/><polyline points={points} fill="none" stroke="var(--vds-color-primary)" strokeWidth="2" vectorEffect="non-scaling-stroke" strokeLinejoin="round" strokeLinecap="round"/></svg><div className="mt-2 grid grid-cols-12 text-center">{data.map((point,index)=><span key={`${point.label}-${index}`} className="text-[10px] text-vds-subtle">{point.label}</span>)}</div></div></section>}
