"use client";
import { configurationModules,permissionKeys } from "../config/defaults";import type { ConfigurationState,PermissionKey } from "../types";import { Header } from "./PipelineEditor";
export function PermissionMatrix({state,onChange}:{state:ConfigurationState;onChange:(state:ConfigurationState)=>void}){return <section><Header title="Roles and permissions" text="Editable policy intent for future governed persistence; current authorization remains unchanged."/>{state.permissions.map(role=><div key={role.role} className="mt-5 overflow-x-auto rounded-3xl border border-vds-border/[.08]"><table className="w-full min-w-[900px] text-xs"><caption className="bg-vds-surface/[.04] p-4 text-left text-base font-semibold">{role.role}</caption><thead><tr className="text-vds-muted"><th className="p-3 text-left">Module</th>{permissionKeys.map(permission=><th key={permission} className="p-3 capitalize">{permission}</th>)}</tr></thead><tbody>{configurationModules.map(module=><tr key={module} className="border-t border-vds-border/[.06]"><th className="p-3 text-left font-medium">{module}</th>{permissionKeys.map(permission=><td key={permission} className="p-3 text-center"><input aria-label={`${role.role} ${module} ${permission}`} type="checkbox" checked={role.modules[module]?.[permission]??false} onChange={event=>onChange({...state,permissions:state.permissions.map(item=>item.role===role.role?{...item,modules:{...item.modules,[module]:{...item.modules[module],[permission as PermissionKey]:event.target.checked}}}:item)})}/></td>)}</tr>)}</tbody></table></div>)}</section>}
