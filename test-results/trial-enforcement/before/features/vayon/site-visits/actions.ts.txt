"use server";import{revalidatePath}from"next/cache";import{operationsContext}from"@/features/vayon/operations/services/context";import{SiteVisitLifecycleService}from"./service";
const roles=new Set(["organization_owner","organization_admin","administrator","sales_manager","sales_agent","project_manager"]);async function authorize(){const c=await operationsContext(),{data}=await c.client.from("organization_members").select("roles(code)").eq("organization_id",c.organizationId).eq("status","active").limit(1).maybeSingle(),role=(data as unknown as{roles?:{code?:string}}|null)?.roles?.code;if(!role||!roles.has(role))throw new Error("You do not have permission to manage site visits.")}
export async function mutateVisitAction(form:FormData){await authorize();const action=String(form.get("action"));if(!["confirm","check-in","check-out","complete","cancel","no-show","reschedule","feedback","follow-up","reserve-unit","release-unit"].includes(action))throw new Error("Unsupported visit action.");const payload:Record<string,unknown>={notes:String(form.get("notes")??"")||undefined,rating:Number(form.get("rating")??0)||undefined,arrivalAt:String(form.get("arrivalAt")??"")||undefined,departureAt:String(form.get("departureAt")??"")||undefined};await(await SiteVisitLifecycleService.production()).mutate({visitId:String(form.get("visitId")),expectedVersion:Number(form.get("version")),action,payload});revalidatePath("/vayon/site-visits")}
