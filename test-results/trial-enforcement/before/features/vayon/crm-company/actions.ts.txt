"use server";
import { revalidatePath } from "next/cache";import { redirect } from "next/navigation";import { CrmCompanyService } from "./service";import { companySchema } from "./validation";
const str=(f:FormData,k:string)=>String(f.get(k)??"").trim(),num=(f:FormData,k:string)=>str(f,k)?Number(str(f,k)):undefined;
const parse=(form:FormData)=>companySchema.safeParse({name:str(form,"name"),logoPath:str(form,"logoPath"),industry:str(form,"industry"),website:str(form,"website"),email:str(form,"email"),phone:str(form,"phone"),employees:num(form,"employees"),revenue:num(form,"revenue"),currency:str(form,"currency")||"USD",address:str(form,"address"),country:str(form,"country"),ownerId:str(form,"ownerId"),notes:str(form,"notes")});
const friendly=(e:unknown)=>{const message=e instanceof Error?e.message:String(e);if(/DUPLICATE_COMPANY_DETECTED/i.test(message))return"A company with this name or email already exists. Open the existing company instead of creating a duplicate.";return/permission|policy|rls/i.test(message)?"You do not have permission to manage companies.":"The company could not be saved. Please retry or contact support."};
export async function createCompanyAction(form:FormData){const p=parse(form);if(!p.success)redirect(`/vayon/crm/companies/new?error=${encodeURIComponent(p.error.issues[0]?.message??"Review the company information.")}`);let id="";try{id=await(await CrmCompanyService.production()).create(p.data);revalidatePath("/vayon/crm/companies")}catch(e){redirect(`/vayon/crm/companies/new?error=${encodeURIComponent(friendly(e))}`)}redirect(`/vayon/crm/companies/${id}?success=Company%20created`)}
export async function updateCompanyAction(id:string,form:FormData){const p=parse(form);if(!p.success)redirect(`/vayon/crm/companies/${id}/edit?error=${encodeURIComponent(p.error.issues[0]?.message??"Review the company information.")}`);try{await(await CrmCompanyService.production()).update(id,Number(str(form,"version")),p.data);revalidatePath(`/vayon/crm/companies/${id}`)}catch(e){redirect(`/vayon/crm/companies/${id}/edit?error=${encodeURIComponent(friendly(e))}`)}redirect(`/vayon/crm/companies/${id}?success=Company%20updated`)}
export async function archiveCompanyAction(id:string,form:FormData){try{await(await CrmCompanyService.production()).archive(id,Number(str(form,"version")));revalidatePath("/vayon/crm/companies")}catch(e){redirect(`/vayon/crm/companies/${id}?error=${encodeURIComponent(friendly(e))}`)}redirect("/vayon/crm/companies?success=Company%20archived")}
