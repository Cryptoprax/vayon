"use server";import{revalidatePath}from"next/cache";import{redirect}from"next/navigation";import{taskSchema,meetingSchema,visitSchema}from"../validation";import{TaskService}from"../services/task.service";import{MeetingService}from"../services/meeting.service";import{SiteVisitService}from"../services/site-visit.service";const value=(f:FormData,k:string)=>String(f.get(k)??"");const message=(e:unknown)=>e instanceof Error?e.message:String(e);const target=(path:string,type:"error"|"success",detail:string)=>`${path}?${type}=${encodeURIComponent(detail)}`;function iso(value:string){const date=new Date(value);return Number.isNaN(date.getTime())?value:date.toISOString()}function taskInput(f:FormData){return{title:value(f,"title"),description:value(f,"description")||undefined,priority:value(f,"priority"),status:value(f,"status"),dueAt:iso(value(f,"dueAt")),assignedUserId:value(f,"assignedUserId")||undefined,relatedType:value(f,"relatedType")||undefined,relatedId:value(f,"relatedId")||undefined}}export async function createTaskAction(f:FormData){const p=taskSchema.safeParse(taskInput(f));if(!p.success)redirect(target("/vayon/tasks","error",p.error.issues[0]?.message??"Review the task details."));let error:string|undefined;try{await new TaskService().create(p.data);revalidatePath("/vayon/tasks")}catch(e){error=message(e)}if(error)redirect(target("/vayon/tasks","error",error));redirect(target("/vayon/tasks","success","Task created successfully."))}export async function updateTaskAction(id:string,version:number,f:FormData){const p=taskSchema.safeParse(taskInput(f));if(!p.success)redirect(target("/vayon/tasks","error",p.error.issues[0]?.message??"Review the task details."));let error:string|undefined;try{await new TaskService().update(id,version,p.data);revalidatePath("/vayon/tasks")}catch(e){error=message(e)}if(error)redirect(target("/vayon/tasks","error",error));redirect(target("/vayon/tasks","success","Task updated successfully."))}export async function completeTaskAction(id:string,version:number){let error:string|undefined;try{await new TaskService().complete(id,version);revalidatePath("/vayon/tasks")}catch(e){error=message(e)}if(error)redirect(target("/vayon/tasks","error",error));redirect(target("/vayon/tasks","success","Task completed."))}export async function archiveTaskAction(id:string){let error:string|undefined;try{await new TaskService().archive(id);revalidatePath("/vayon/tasks")}catch(e){error=message(e)}if(error)redirect(target("/vayon/tasks","error",error));redirect(target("/vayon/tasks","success","Task archived."))}export async function scheduleMeetingAction(f:FormData){const p=meetingSchema.safeParse({title:value(f,"title"),meetingType:value(f,"meetingType"),startsAt:iso(value(f,"startsAt")),endsAt:iso(value(f,"endsAt")),relatedType:value(f,"relatedType")||undefined,relatedId:value(f,"relatedId")||undefined});if(!p.success)redirect(target("/vayon/meetings","error",p.error.issues[0]?.message??"Review the meeting details."));let error:string|undefined;try{await new MeetingService().schedule(p.data);revalidatePath("/vayon/meetings")}catch(e){error=message(e)}if(error)redirect(target("/vayon/meetings","error",error));redirect(target("/vayon/meetings","success","Meeting scheduled."))}export async function scheduleVisitAction(f:FormData){const p=visitSchema.safeParse({propertyId:value(f,"propertyId"),leadId:value(f,"leadId")||undefined,assignedAgentId:value(f,"assignedAgentId")||undefined,startsAt:iso(value(f,"startsAt")),status:value(f,"status"),outcome:value(f,"outcome")||undefined,notes:value(f,"notes")||undefined});if(!p.success)redirect(target("/vayon/site-visits","error",p.error.issues[0]?.message??"Review the visit details."));let error:string|undefined;try{await new SiteVisitService().schedule(p.data);revalidatePath("/vayon/site-visits")}catch(e){error=message(e)}if(error)redirect(target("/vayon/site-visits","error",error));redirect(target("/vayon/site-visits","success","Site visit scheduled."))}
