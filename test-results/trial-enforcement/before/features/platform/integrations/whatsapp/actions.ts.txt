"use server";import{revalidatePath}from"next/cache";import{operationsContext}from"@/features/vayon/operations/services/context";import{TokenCryptoService}from"../google/services/token-crypto.service";export async function connectWhatsAppAction(form:FormData){const phoneNumberId=String(form.get("phoneNumberId")??""),businessAccountId=String(form.get("businessAccountId")??""),token=String(form.get("accessToken")??"");if(!phoneNumberId||!businessAccountId||!token)throw new Error("WhatsApp Business credentials are required.");const c=await operationsContext(),encrypted=new TokenCryptoService().encrypt(token),{error}=await c.client.rpc("connect_whatsapp",{p_workspace_id:c.workspaceId,p_phone_number_id:phoneNumberId,p_business_account_id:businessAccountId,p_token_ciphertext:encrypted.ciphertext,p_token_iv:encrypted.iv,p_token_tag:encrypted.tag});if(error)throw error;revalidatePath("/vayon/settings/integrations")}export async function disconnectWhatsAppAction(){const c=await operationsContext(),{error}=await c.client.rpc("disconnect_whatsapp",{p_workspace_id:c.workspaceId});if(error)throw error;revalidatePath("/vayon/settings/integrations")}
