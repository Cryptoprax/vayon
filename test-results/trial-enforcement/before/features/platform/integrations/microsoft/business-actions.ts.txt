"use server";import{revalidatePath}from"next/cache";import{redirect}from"next/navigation";import{OutlookMailService}from"./services/outlook.service";import{OutlookCalendarService}from"./services/outlook-calendar.service";import{OneDriveService}from"./services/onedrive.service";const value=(form:FormData,key:string)=>String(form.get(key)??"").trim();export async function composeOutlookAction(form:FormData){const input={to:value(form,"to").split(",").map(item=>item.trim()).filter(Boolean),subject:value(form,"subject"),body:value(form,"body")};if(!input.to.length||!input.subject)redirect("/vayon/communications/outlook?error=Recipient%20and%20subject%20are%20required");if(value(form,"mode")==="draft")await new OutlookMailService().saveDraft(input);else await new OutlookMailService().send(input);revalidatePath("/vayon/communications/outlook");redirect("/vayon/communications/outlook?success=Outlook%20request%20completed")}export async function createOutlookMeetingAction(form:FormData){const start=value(form,"start"),end=value(form,"end"),title=value(form,"title"),timeZone=value(form,"timeZone")||"UTC";if(!title||!start||!end)redirect("/vayon/calendar/outlook?error=Title%20and%20times%20are%20required");await new OutlookCalendarService().create({calendarId:"primary",title,start:new Date(start).toISOString(),end:new Date(end).toISOString(),timeZone,attendees:value(form,"guests").split(",").map(item=>item.trim()).filter(Boolean),recurrence:value(form,"recurring")?["RRULE:FREQ=DAILY"]:[]});revalidatePath("/vayon/calendar/outlook");redirect("/vayon/calendar/outlook?success=Outlook%20meeting%20created")}export async function uploadOneDriveAction(form:FormData){const file=form.get("file");if(!(file instanceof File)||!file.size)redirect("/vayon/documents/onedrive?error=Choose%20a%20file");await new OneDriveService().upload(file,value(form,"folderId")||undefined);revalidatePath("/vayon/documents/onedrive");redirect("/vayon/documents/onedrive?success=OneDrive%20upload%20completed")}
