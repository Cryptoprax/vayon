"use server";import{revalidatePath}from"next/cache";import{StorageService,type AssetKind}from"./storage.service";const kinds=new Set<AssetKind>(["property-images","property-documents","lead-documents","deal-documents","company-logo","profile-photo"]);export async function uploadAssetAction(form:FormData){const kind=String(form.get("kind"))as AssetKind,entityId=String(form.get("entityId")??""),file=form.get("file");if(!kinds.has(kind)||!entityId||!(file instanceof File)||file.size===0)throw new Error("A valid asset is required.");if(file.size>20*1024*1024)throw new Error("Asset exceeds the 20 MB limit.");await new StorageService().upload(kind,entityId,file);revalidatePath("/vayon/storage")}
