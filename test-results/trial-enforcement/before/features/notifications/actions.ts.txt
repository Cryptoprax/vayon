"use server";import{revalidatePath}from"next/cache";import{NotificationService}from"./services/notification.service";import{notificationCategories,type NotificationCategory,type NotificationPreference}from"./types";
const refresh=()=>{revalidatePath("/vayon/notifications");revalidatePath("/vayon/notifications/inbox");revalidatePath("/vayon/notifications/history");revalidatePath("/vayon/notifications/preferences")};
export async function mutateNotificationAction(form:FormData){const action=String(form.get("action"))as"read"|"unread"|"archive"|"restore"|"star"|"unstar"|"snooze";if(!["read","unread","archive","restore","star","unstar","snooze"].includes(action))throw new Error("Unsupported notification action.");await new NotificationService().mutate(String(form.get("notificationId")??""),action,action==="snooze"?new Date(Date.now()+Number(form.get("minutes")??60)*60000).toISOString():undefined);refresh()}
export async function dismissNotificationAction(form:FormData){form.set("action","archive");return mutateNotificationAction(form)}
export async function saveNotificationPreferencesAction(form:FormData){const categories=Object.fromEntries(notificationCategories.map(x=>[x,form.get(`category:${x}`)==="on"]))as Partial<Record<NotificationCategory,boolean>>;const value:NotificationPreference={email:form.get("email")==="on",inApp:form.get("inApp")==="on",browserPush:form.get("browserPush")==="on",whatsapp:form.get("whatsapp")==="on",webhook:form.get("webhook")==="on",muted:form.get("muted")==="on",quietHoursStart:String(form.get("quietHoursStart")||"")||undefined,quietHoursEnd:String(form.get("quietHoursEnd")||"")||undefined,digestFrequency:String(form.get("digestFrequency")||"instant")as NotificationPreference["digestFrequency"],categories};await new NotificationService().savePreferences(value);refresh()}
export async function scheduleReminderAction(form:FormData){await new NotificationService().scheduleReminder({kind:String(form.get("kind"))as Parameters<NotificationService["scheduleReminder"]>[0]["kind"],title:String(form.get("title")??""),remindAt:new Date(String(form.get("remindAt")??"")).toISOString()});refresh()}
